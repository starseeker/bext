if (ENABLE_RE2C)

  git_submodule_init(re2c CMakeLists.txt)

  RegisterDeps(RE2C)

  ExternalProject_Add(RE2C_BLD
    BEXT_BUILD_ROLE HOST
    URL "${CMAKE_CURRENT_SOURCE_DIR}/re2c"
    BUILD_ALWAYS ${EXT_BUILD_ALWAYS} ${LOG_OPTS}
    PATCH_COMMAND ${PATCH_EXECUTABLE};-E;-p1;${PATCH_OPTIONS};-i;${CMAKE_CURRENT_SOURCE_DIR}/re2c.patch
    CMAKE_ARGS
    ${BUILD_TYPE_SPECIFIER}
    -DRE2C_BUILD_RE2GO=OFF
    -DRE2C_BUILD_RE2RUST=OFF
    -DRE2C_REBUILD_DOCS=OFF
    -DRE2C_BUILD_RE2D=OFF
    -DRE2C_BUILD_RE2GO=OFF
    -DRE2C_BUILD_RE2HS=OFF
    -DRE2C_BUILD_RE2JAVA=OFF
    -DRE2C_BUILD_RE2JS=OFF
    -DRE2C_BUILD_RE2OCAML=OFF
    -DRE2C_BUILD_RE2PY=OFF
    -DRE2C_BUILD_RE2RUST=OFF
    -DRE2C_BUILD_RE2SWIFT=OFF
    -DRE2C_BUILD_RE2V=OFF
    -DRE2C_BUILD_RE2ZIG=OFF
    -DBIN_DIR=${BIN_DIR}
    -DLIB_DIR=${LIB_DIR}
    -DDATA_DIR=${DATA_DIR}
    -DBUILD_STATIC_LIBS=${BUILD_STATIC_LIBS}
    -DCMAKE_CXX_COMPILER=${BEXT_HOST_CXX_COMPILER}
    -DCMAKE_C_COMPILER=${BEXT_HOST_C_COMPILER}
    -DCMAKE_INSTALL_PREFIX=${BEXT_HOST_INSTALL_PREFIX}
    -DCMAKE_INSTALL_LIBDIR:PATH=${LIB_DIR}
    -DCMAKE_INSTALL_RPATH=${BEXT_HOST_INSTALL_PREFIX}/${LIB_DIR}
    LOG_CONFIGURE ${EXT_BUILD_QUIET}
    LOG_BUILD ${EXT_BUILD_QUIET}
    LOG_INSTALL ${EXT_BUILD_QUIET}
    LOG_OUTPUT_ON_FAILURE ${EXT_BUILD_QUIET}
    STEP_TARGETS install
    )

  TargetInstallDeps(RE2C RE2C_DEPENDS)

endif (ENABLE_RE2C)
