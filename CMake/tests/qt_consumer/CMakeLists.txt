cmake_minimum_required(VERSION 3.26)
project(BextQtConsumer LANGUAGES CXX)
if(NOT BEXT_QT_PREFIX)
  message(FATAL_ERROR "Set BEXT_QT_PREFIX to the target Qt installation")
endif()
list(APPEND CMAKE_FIND_ROOT_PATH "${BEXT_QT_PREFIX}")
list(PREPEND CMAKE_PREFIX_PATH "${BEXT_QT_PREFIX}")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Svg OpenGL OpenGLWidgets Network)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
add_executable(qt_consumer consumer.cpp form.ui fixture.qrc)
target_compile_features(qt_consumer PRIVATE cxx_std_17)
target_link_libraries(qt_consumer PRIVATE
  Qt6::Widgets Qt6::Svg Qt6::OpenGLWidgets Qt6::Network)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/selected-tools-$<CONFIG>.txt"
  CONTENT "moc=$<TARGET_FILE:Qt6::moc>\nuic=$<TARGET_FILE:Qt6::uic>\nrcc=$<TARGET_FILE:Qt6::rcc>\n")
enable_testing()
add_test(NAME qt_generated_sources COMMAND qt_consumer)
if(CMAKE_CROSSCOMPILING AND NOT CMAKE_CROSSCOMPILING_EMULATOR)
  set_tests_properties(qt_generated_sources PROPERTIES DISABLED TRUE)
endif()
set_tests_properties(qt_generated_sources PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
