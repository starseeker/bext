cmake_minimum_required(VERSION 3.26)
project(BextOpenNurbsConsumer LANGUAGES CXX)
if (NOT BEXT_OPENNURBS_PREFIX)
  message(FATAL_ERROR "Set BEXT_OPENNURBS_PREFIX to the target installation")
endif ()
list(APPEND CMAKE_FIND_ROOT_PATH "${BEXT_OPENNURBS_PREFIX}")
find_path(opennurbs_include opennurbs.h
  PATHS "${BEXT_OPENNURBS_PREFIX}/include/OpenNURBS" NO_DEFAULT_PATH REQUIRED)
find_library(opennurbs_library NAMES OpenNURBS
  PATHS "${BEXT_OPENNURBS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(opennurbs_consumer consumer.cpp)
target_compile_features(opennurbs_consumer PRIVATE cxx_std_17)
target_compile_definitions(opennurbs_consumer PRIVATE OPENNURBS_IMPORTS)
target_include_directories(opennurbs_consumer PRIVATE "${opennurbs_include}")
target_link_libraries(opennurbs_consumer PRIVATE "${opennurbs_library}")
enable_testing()
add_test(NAME opennurbs_consumer COMMAND opennurbs_consumer)
if (CMAKE_CROSSCOMPILING AND NOT CMAKE_CROSSCOMPILING_EMULATOR)
  set_tests_properties(opennurbs_consumer PROPERTIES DISABLED TRUE)
endif ()
if (WIN32 AND NOT CMAKE_CROSSCOMPILING)
  set_tests_properties(opennurbs_consumer PROPERTIES ENVIRONMENT_MODIFICATION
    "PATH=path_list_prepend:${BEXT_OPENNURBS_PREFIX}/bin")
endif ()
