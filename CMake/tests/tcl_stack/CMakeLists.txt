cmake_minimum_required(VERSION 3.26)
project(BextTclStackConsumer LANGUAGES C)
if (NOT BEXT_TCL_PREFIX)
  message(FATAL_ERROR "Set BEXT_TCL_PREFIX to the target Tcl/Tk/Itcl/Itk installation")
endif ()
list(APPEND CMAKE_FIND_ROOT_PATH "${BEXT_TCL_PREFIX}")
find_path(stack_include tcl.h PATHS "${BEXT_TCL_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(tcl_stub NAMES tclstub86 tclstub8.6
  PATHS "${BEXT_TCL_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(tk_stub NAMES tkstub86 tkstub8.6
  PATHS "${BEXT_TCL_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
foreach(package itcl itk)
  string(TOUPPER "${package}" upper)
  file(STRINGS "${stack_include}/${package}.h" version_line
    REGEX "^#define[ \t]+${upper}_PATCH_LEVEL[ \t]+")
  if (NOT version_line MATCHES "\"([0-9.]+)\"")
    message(FATAL_ERROR "Missing ${upper}_PATCH_LEVEL in installed ${package}.h")
  endif ()
  set(version "${CMAKE_MATCH_1}")
  string(REPLACE "." "" compact "${version}")
  find_library(${package}_stub NAMES ${package}stub${compact} ${package}stub${version}
    PATHS "${BEXT_TCL_PREFIX}/lib/${package}${version}" NO_DEFAULT_PATH REQUIRED)
endforeach ()
add_library(bext_tcl_stack MODULE stack.c)
target_include_directories(bext_tcl_stack PRIVATE "${stack_include}")
if (UNIX AND NOT APPLE)
  find_package(X11 REQUIRED)
  target_include_directories(bext_tcl_stack PRIVATE "${X11_INCLUDE_DIR}")
endif ()
target_compile_definitions(bext_tcl_stack PRIVATE USE_TCL_STUBS USE_TK_STUBS USE_ITCL_STUBS USE_ITK_STUBS)
target_link_libraries(bext_tcl_stack PRIVATE "${itk_stub}" "${itcl_stub}" "${tk_stub}" "${tcl_stub}")
if (CMAKE_SYSTEM_NAME MATCHES "^(Linux|FreeBSD)$")
  target_link_options(bext_tcl_stack PRIVATE "LINKER:--no-undefined")
endif ()
