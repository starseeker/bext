cmake_minimum_required(VERSION 3.26)
project(BextQtHostProbe LANGUAGES CXX)
if(NOT BEXT_HOST_QT_PATH OR NOT BEXT_QT_VERSION)
  message(FATAL_ERROR "Set BEXT_HOST_QT_PATH and BEXT_QT_VERSION")
endif()
set(CMAKE_PREFIX_PATH "${BEXT_HOST_QT_PATH}")
# Recheck when the selected host installation changes on a later configure.
foreach(package Qt6 Qt6CoreTools Qt6GuiTools Qt6WidgetsTools Qt6DBusTools)
  unset(${package}_DIR CACHE)
endforeach()
find_package(Qt6 ${BEXT_QT_VERSION} EXACT REQUIRED
  COMPONENTS CoreTools GuiTools WidgetsTools DBusTools
  PATHS "${BEXT_HOST_QT_PATH}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
foreach(tool moc rcc uic)
  get_target_property(location Qt6::${tool} IMPORTED_LOCATION)
  if(NOT location)
    get_target_property(configs Qt6::${tool} IMPORTED_CONFIGURATIONS)
    foreach(config IN LISTS configs)
      get_target_property(location Qt6::${tool} IMPORTED_LOCATION_${config})
      if(location)
        break()
      endif()
    endforeach()
  endif()
  execute_process(COMMAND "${location}" -v TIMEOUT 10
    RESULT_VARIABLE result OUTPUT_VARIABLE output ERROR_VARIABLE error)
  string(REGEX MATCH "[0-9]+[.][0-9]+[.][0-9]+" version "${output}${error}")
  if(NOT result EQUAL 0 OR NOT version VERSION_EQUAL BEXT_QT_VERSION)
    message(FATAL_ERROR "Native ${tool} must run Qt ${BEXT_QT_VERSION}: ${result}\n${output}${error}")
  endif()
  message(STATUS "Native ${tool} ${version}: ${location}")
endforeach()
